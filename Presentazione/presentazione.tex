% SLIDE 1: LINEARIZZAZIONE DEL MODELLO A ORDINE RIDOTTO (ROM)
% -------------------------------------------------------------
\begin{frame}{Linearizzazione del modello del paper}
\vspace{0.2em}
{\small Sviluppo delle forze attorno all'equilibrio}
\vspace{0.2em}
\[
  \frac{f_{DD}^{(1)}}{\varepsilon_0} \approx c_{0,DD}^{(1)} + c_{1,DD}^{(1)} q_1, \qquad
  \frac{f_{DA}^{(1)}}{\varepsilon_0} \approx c_{0,DA}^{(1)}, \qquad
  \beta_1 \approx c_{1,\beta}^{(1)} q_1
\]

\vspace{0.6em}
{\small Euqazioni del modello del MEMS}
\vspace{0.2em}
\[
  \begin{cases}
    \ddot{q}_1 + \dfrac{\omega_{01}}{Q_1}\dot{q}_1 + \left(c_{1,\beta}^{(1)} - \varepsilon_0 c_{1,DD}^{(1)} V_{DC}^2\right) q_1 \approx \varepsilon_0 c_{0,DD}^{(1)} V_{DC}^2 + 2\varepsilon_0 c_{0,DA}^{(1)} V_{DC}\, v_{in}(t) \\[10pt]
    i_s(t) \approx \varepsilon_0 c_{0,DA}^{(1)} V_{DC}\, \dot{q}_1(t)
  \end{cases}
\]
\end{frame}

% -------------------------------------------------------------
% SLIDE 2: DINAMICA LIBERA E CIRCUITO EQUIVALENTE RLC
% -------------------------------------------------------------
\begin{frame}{Dinamica libera e circuito equivalente RLC}
\vspace{0.2em}
{\small Risposta libera di ringdown (allo spegnimento di $v_{in}$):}
\vspace{0.2em}
\[
  q_{1}(t) = A_0 e^{-t/\tau} \cos(\omega_r t + \phi_0)
\]
\[
  \lambda_{1,2} = \mu \pm j\omega_r, \qquad \mu = -\frac{\omega_{01}}{2Q_1} = -\frac{1}{\tau}
\]
\[
  \omega_r = \sqrt{c_{1,\beta}^{(1)} - \varepsilon_0 c_{1,DD}^{(1)} V_{DC}^2 - \frac{\omega_{01}^2}{4Q_1^2}} \approx \sqrt{\omega_{01}^2 - \varepsilon_0 c_{1,DD}^{(1)} V_{DC}^2}
\]

\vspace{0.4em}
{\small Parametri del ramo motazionale RLC serie equivalente:}
\vspace{0.2em}
\[
  L_m = \frac{1}{2(\varepsilon_0 c_{0,DA}^{(1)} V_{DC})^2}, \qquad
  R_m = \frac{\omega_{01} L_m}{Q_1}, \qquad
  C_m = \frac{1}{\omega_1^2(V_{DC}) L_m}
\]
\end{frame}

% SLIDE 8: ANALISI E TARATURA DEL CIRCUITO DI COMPENSAZIONE
% -------------------------------------------------------------
\begin{frame}{Analisi e taratura del circuito di compensazione}
\footnotesize
\vspace{-0.5em}
\begin{columns}[t]
  \begin{column}{0.48\textwidth}
    Funzione di trasferimento di $U_1$:
    \vspace{0.4em}
    \begin{equation*}
      T_1(s) = -\frac{s C_{\phi} R_f}{1 + s C_{\phi} R_{in}}
    \end{equation*}
    \vspace{0.4em}
    con $R_{in} = R_{\phi 1} + R_{P\phi 1}$ e $R_f = R_{\phi 2} + R_{P\phi 2}$.
    \begin{itemize}
      \setlength{\itemsep}{0.4em}
      \item Modulo: $|T_1(\omega)| = \dfrac{\omega C_\phi R_f}{\sqrt{1 + (\omega C_\phi R_{in})^2}}$
      \item Fase: $\angle T_1(\omega) = 180^\circ + \arctan\left(\dfrac{1}{\omega C_\phi R_{in}}\right)$
    \end{itemize}
  \end{column}

  \begin{column}{0.48\textwidth}
    Procedura di taratura:
    \begin{enumerate}
      \setlength{\itemsep}{0.4em}
      \item Facciamo variare $R_{P\phi 1}$ per ottenere $\Delta\phi = 180.0^\circ$ alla frequenza di risonanza.
      \item Facciamo variare $R_{P\phi 2}$ per avere guadagno unitario $|T_1(f_0)| = 1$ ($T_1 = -1$).\\
      \vspace{0.4em}
      Poiché $\dfrac{\partial \angle T_1}{\partial R_{P\phi 2}} = 0$, la fase non viene alterata.
    \end{enumerate}
  \end{column}
\end{columns}
\end{frame}

% SLIDE 17: ORIGINE DEL DISTURBO
% -------------------------------------------------------------
\begin{frame}{Origine del disturbo}
\begin{itemize}
  \setlength{\itemsep}{0.02em}
  \item Con $V_{DC} = 0\text{ V}$:
  \[
    i_{\text{RLC}}(t) \approx c_{0,DA}^{(1)} V_{DC} \dot{q}_1(t) = 0\text{ A}
  \]

  \item Corrente parassita netta iniettata al nodo d'ingresso del TIA dovuta a $\Delta C = C_p - C_c$:
  \[
    i_{\text{net}}(t) = \Delta C \frac{dv_{in}(t)}{dt}
  \]

  \item Gradino di corrente allo spegnimento del gate:
  \[
    \Delta I = 0 - i_{\text{net}}(t) = -\Delta C \frac{dv_{in}}{dt}
  \]
  \begin{itemize}
    \setlength{\itemsep}{0.1em}
    \item Pendenza negativa $\frac{dv_{in}}{dt} < 0 \implies \Delta I > 0$: picco iniziale positivo di $+804\text{ mV}$ 
    \item Pendenza positiva $\frac{dv_{in}}{dt} > 0 \implies \Delta I < 0$: picco iniziale negativo di $-607\text{ mV}$
  \end{itemize}


    \item Oscillazione parassita a $\approx 600\text{ kHz}$
    \item Scende sotto il $10\%$ del picco massimo in $\approx 2.2\ \mu\text{s}$
  \end{itemize}
\end{frame}
